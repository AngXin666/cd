/**
 * 格式化时间为 HH:mm 格式
 * @param timeStr - 时间字符串或 Date 对象
 * @returns 格式化后的时间字符串
 */
export function formatTime(timeStr: string | Date): string {
  const date = typeof timeStr === 'string' ? new Date(timeStr) : timeStr
  const hours = String(date.getHours()).padStart(2, '0')
  const minutes = String(date.getMinutes()).padStart(2, '0')
  return `${hours}:${minutes}`
}

/**
 * 格式化时间为 HH:mm:ss 格式
 * @param timeStr - 时间字符串或 Date 对象
 * @returns 格式化后的时间字符串
 */
export function formatTimeWithSeconds(timeStr: string | Date): string {
  const date = typeof timeStr === 'string' ? new Date(timeStr) : timeStr
  const hours = String(date.getHours()).padStart(2, '0')
  const minutes = String(date.getMinutes()).padStart(2, '0')
  const seconds = String(date.getSeconds()).padStart(2, '0')
  return `${hours}:${minutes}:${seconds}`
}
```

/**
 * 格式化日期为 YYYY-MM-DD 格式
 */
export function formatDateISO(dateStr: string | Date): string {
  const date = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  return `${year}-${month}-${day}`
}

/**
 * 格式化日期为 MM/DD 周X 格式
 */
export function formatDateWithWeekday(dateStr: string): string {
  const date = new Date(dateStr)
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  const weekDays = ['日', '一', '二', '三', '四', '五', '六']
  const weekDay = weekDays[date.getDay()]
  return `${month}/${day} 周${weekDay}`
}

/**
 * 格式化日期为 M月D日 格式
 */
export function formatDateChinese(dateStr: string): string {
  const date = new Date(dateStr)
  return `${date.getMonth() + 1}月${date.getDate()}日`
}

/**
 * 格式化日期时间为 YYYY-MM-DD HH:mm 格式
 */
export function formatDateTime(dateStr: string): string {
  const date = new Date(dateStr)
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  const hours = String(date.getHours()).padStart(2, '0')
  const minutes = String(date.getMinutes()).padStart(2, '0')
  return `${year}-${month}-${day} ${hours}:${minutes}`
}
```

 * 平台检测与兼容工具
 * @module utils/platform
 */

/**
 * 检测当前是否为 H5 环境
 */
export const isH5 = process.env.TARO_ENV === 'h5'

/**
 * 检测当前是否为微信小程序环境
 */
export const isWeapp = process.env.TARO_ENV === 'weapp'

/**
 * 兼容 H5 和小程序的 showLoading
 * H5 环境下不显示原生 loading（可集成自定义组件）
 */
export function showLoading(options: { title: string }): void {
  if (!isH5) {
    Taro.showLoading(options)
  }
  // H5 环境可以在这里集成自定义 loading 组件
}

/**
 * 兼容 H5 和小程序的 hideLoading
 */
export function hideLoading(): void {
  if (!isH5) {
    Taro.hideLoading()
  }
}

/**
 * 兼容 H5 和小程序的 showToast
 */
export function showToast(options: Taro.showToast.Option): void {
  if (isH5) {
    // H5 环境使用简化版本
    Taro.showToast({
      ...options,
      duration: options.duration || 2000
    })
  } else {
    Taro.showToast(options)
  }
}

### 6. API 层 Supabase 查询模式

**问题描述**：`src/db/api/` 目录下多个文件都有类似的 Supabase 查询封装模式。

**观察到的模式**：
```typescript
// 常见的查询模式
const { data, error } = await supabase
  .from('table_name')
  .select('*')
  .eq('field', value)
  .order('created_at', { ascending: false })

if (error) {
  console.error('操作失败:', error)
  return null // 或 []
}
return data
```

**重构建议**：
可以考虑创建通用的查询辅助函数，但需要谨慎：
- 每个 API 的业务逻辑不同
- 错误处理需求不同
- 过度抽象可能降低可读性

**建议方案**：保持现状，或仅抽取最通用的部分：

```typescript
// src/db/helpers.ts 中添加
export async function safeQuery<T>(
  queryFn: () => Promise<{ data: T | null; error: any }>,
  errorMessage: string
): Promise<T | null> {
  try {
    const { data, error } = await queryFn()
    if (error) {
      console.error(errorMessage, error)
      return null
    }
    return data
  } catch (error) {
    console.error(errorMessage, error)
    return null
  }
}
```
